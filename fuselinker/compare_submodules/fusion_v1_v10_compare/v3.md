```python
class FusionV3(nn.Module):
    def __init__(self, hidden_dim):
        super().__init__()

        self.gate = nn.Linear(hidden_dim * 2, 1)

    def forward(self, text_x, domain_x):
        gate = torch.sigmoid(
            self.gate(torch.cat([text_x, domain_x], dim=-1))
        )

        return gate * text_x + (1.0 - gate) * domain_x
```
