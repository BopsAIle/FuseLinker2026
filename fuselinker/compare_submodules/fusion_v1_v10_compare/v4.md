```python
class FusionV4(nn.Module):
    def __init__(self, hidden_dim):
        super().__init__()

        self.gate = nn.Linear(hidden_dim, hidden_dim)

    def forward(self, text_x, domain_x):
        diff = torch.abs(text_x - domain_x)

        gate = torch.sigmoid(
            self.gate(diff)
        )

        return gate * text_x + (1.0 - gate) * domain_x
```
